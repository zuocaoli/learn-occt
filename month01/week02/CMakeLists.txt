cmake_minimum_required(VERSION 4.3)
set(CMAKE_EXPERIMENTAL_CXX_IMPORT_STD "451f2fe2-a8a2-47c3-bc32-94786d8fc91b")
project(geometry_month01_week02 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_MODULE_STD ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR CMAKE_CXX_COMPILER_VERSION VERSION_LESS 22)
    message(FATAL_ERROR "Use Clang 22+ and the default preset.")
endif()
if(NOT 23 IN_LIST CMAKE_CXX_COMPILER_IMPORT_STD)
    message(FATAL_ERROR "Install matching clang-scan-deps and libstdc++ module development files for import std.")
endif()

option(GEOMETRY_BUILD_GUI "Build Qt Widgets / OCCT viewer" ON)
if(NOT OpenCASCADE_DIR)
    set(OpenCASCADE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../occt/lib/cmake/opencascade" CACHE PATH "OCCT CMake package")
endif()
set(occt_components FoundationClasses ModelingData ModelingAlgorithms)
if(GEOMETRY_BUILD_GUI)
    list(APPEND occt_components Visualization)
endif()
find_package(OpenCASCADE CONFIG REQUIRED COMPONENTS ${occt_components})
message(STATUS "OCCT: ${OpenCASCADE_MAJOR_VERSION}.${OpenCASCADE_MINOR_VERSION}.${OpenCASCADE_MAINTENANCE_VERSION} ${OpenCASCADE_DEVELOPMENT_VERSION}")
configure_file(framework/runtime_config.hpp.in generated/runtime_config.hpp @ONLY)

add_library(geometry_core STATIC
    framework/experiment.cpp framework/inspect.cpp framework/artifacts.cpp
    framework/command_line.cpp experiments/box_transform.cpp experiments/course.cpp)
target_include_directories(geometry_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(geometry_core PUBLIC TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim)
target_compile_options(geometry_core PRIVATE -Wall -Wextra -Wpedantic)

# The supplied OCCT DSOs have no RUNPATH. DT_RPATH also resolves their indirect dependencies.
function(geometry_runtime target)
    set_target_properties(${target} PROPERTIES BUILD_RPATH "${OpenCASCADE_LIBRARY_DIR}")
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_options(${target} PRIVATE "LINKER:--disable-new-dtags")
    endif()
endfunction()

add_executable(console-demo console.cpp)
target_link_libraries(console-demo PRIVATE geometry_core)
geometry_runtime(console-demo)

if(GEOMETRY_BUILD_GUI)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The initial viewer uses Linux X11/XWayland. Disable GEOMETRY_BUILD_GUI for the core.")
    endif()
    find_package(Qt6 6.11 REQUIRED COMPONENTS Widgets)
    find_package(X11 REQUIRED)
    add_library(geometry_ui STATIC framework/ui/viewport.cpp framework/ui/workbench.cpp)
    set_target_properties(geometry_ui PROPERTIES AUTOMOC ON)
    target_compile_definitions(geometry_ui PUBLIC QT_NO_KEYWORDS)
    target_link_libraries(geometry_ui PUBLIC geometry_core Qt6::Widgets TKService TKV3d TKOpenGl X11::X11)
    target_compile_options(geometry_ui PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(demo main.cpp)
    target_link_libraries(demo PRIVATE geometry_ui)
    geometry_runtime(demo)
else()
    add_executable(demo console.cpp)
    target_link_libraries(demo PRIVATE geometry_core)
    geometry_runtime(demo)
endif()
